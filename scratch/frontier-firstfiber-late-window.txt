 13223	private theorem exists_second_outside_two_pairs
 13224	    {α : Type*} [DecidableEq α]
 13225	    (K : Finset α) (q a b c d : α)
 13226	    (hKcard : K.card = 4)
 13227	    (hqOutside : q ∉ ({a, b, c, d} : Finset α))
 13228	    (hfirst : ¬ (a ∈ K ∧ b ∈ K))
 13229	    (hsecond : ¬ (c ∈ K ∧ d ∈ K)) :
 13230	    ∃ v ∈ K, v ≠ q ∧ v ∉ ({a, b, c, d} : Finset α) := by
 13231	  let E : Finset α := ({a, b, c, d} : Finset α)
 13232	  have hfirstCard : (K ∩ ({a, b} : Finset α)).card ≤ 1 := by
 13233	    rw [Finset.card_le_one]
 13234	    intro x hx y hy
 13235	    simp only [Finset.mem_inter, Finset.mem_insert,
 13236	      Finset.mem_singleton] at hx hy
 13237	    rcases hx.2 with rfl | rfl <;>
 13238	      rcases hy.2 with rfl | rfl
 13239	    · rfl
 13240	    · exact False.elim (hfirst ⟨hx.1, hy.1⟩)
 13241	    · exact False.elim (hfirst ⟨hy.1, hx.1⟩)
 13242	    · rfl
 13243	  have hsecondCard : (K ∩ ({c, d} : Finset α)).card ≤ 1 := by
 13244	    rw [Finset.card_le_one]
 13245	    intro x hx y hy
 13246	    simp only [Finset.mem_inter, Finset.mem_insert,
 13247	      Finset.mem_singleton] at hx hy
 13248	    rcases hx.2 with rfl | rfl <;>
 13249	      rcases hy.2 with rfl | rfl
 13250	    · rfl
 13251	    · exact False.elim (hsecond ⟨hx.1, hy.1⟩)
 13252	    · exact False.elim (hsecond ⟨hy.1, hx.1⟩)
 13253	    · rfl
 13254	  have hinterEq :
 13255	      K ∩ E =
 13256	        (K ∩ ({a, b} : Finset α)) ∪
 13257	          (K ∩ ({c, d} : Finset α)) := by
 13258	    ext x
 13259	    simp only [E, Finset.mem_inter, Finset.mem_union,
 13260	      Finset.mem_insert, Finset.mem_singleton]
 13261	    tauto

 13263	    rw [hinterEq]

 13265	      ((K ∩ {a, b}) ∪ (K ∩ {c, d})).card
 13266	          ≤ (K ∩ {a, b}).card + (K ∩ {c, d}).card :=
 13267	        Finset.card_union_le _ _

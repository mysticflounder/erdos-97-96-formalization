  exact ⟨{

    source₂ := x
    source₁_mem_A := P.source₁_mem_A
    source₂_mem_A := hxA

    source₂_mem_capInterior := hxInterior

    directed_omission := Or.inl ⟨hxOutside,
      (cross_deletion_survives_iff_not_mem_selected_support H
        P.source₁_mem_A).mpr hxOutside⟩ }⟩

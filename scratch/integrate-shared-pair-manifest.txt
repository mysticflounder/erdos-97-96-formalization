
def _active_counts(bank: Any) -> Counter[str]:
    return Counter(
        str(certificate["stage"]) for certificate in bank.active.values()
    )


def _artifact_hashes(out: Path) -> dict[str, str]:
    names = (
        "base.cnf",
        "smoke.json",
        "bootstrap.json",
        "learned-certificates.jsonl",
        "survivors.jsonl",
        "solver-logs.jsonl",
        "cube-batches.jsonl",
        "failure.json",
        "terminal.cnf",
        "terminal.drat",
        "terminal.drat.check",
    )
    return {
        name: _sha256_file(out / name)
        for name in names
        if (out / name).is_file()
    }
